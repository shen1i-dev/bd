SELECT table_name
FROM information_schema.tables
WHERE table_schema = 'public'
ORDER BY table_name;

#### And or not
-- №1. AND
-- Товари дорожчі за 15000 грн і дешевші за 50000 грн
SELECT *
FROM products
WHERE unit_price > 15000
  AND unit_price < 50000;
![alt text](image-20.png)

-- №2. AND + OR
-- Юридичні клієнти з Києва або Львова
SELECT *
FROM customers
WHERE (city = 'Київ' OR city = 'Львів')
  AND customer_type = 'company';
![alt text](image-21.png)

-- №3. AND + NOT
-- Товари, які є в наявності та не зняті з продажу
SELECT *
FROM products
WHERE units_in_stock > 0
  AND NOT discontinued;
![alt text](image-22.png)

-- №4. AND + OR
-- Клієнти з Києва або Львова, які мають email
SELECT *
FROM customers
WHERE (city = 'Київ' OR city = 'Львів')
  AND email IS NOT NULL;
![alt text](image-23.png)

-- №5. OR
-- Замовлення, які доставлені або вже мають дату відправлення
SELECT *
FROM orders
WHERE order_status = 'delivered'
   OR shipped_date IS NOT NULL;
![alt text](image-24.png)

-- №6. AND
-- Працівники, у яких вказані керівник і номер телефону
SELECT *
FROM employees
WHERE reports_to IS NOT NULL
  AND phone IS NOT NULL;
![alt text](image-25.png)

#### In between null not null
-- №1. IN
-- Клієнти з Києва, Харкова, Одеси або Дніпра
SELECT *
FROM customers
WHERE city IN ('Київ', 'Харків', 'Одеса', 'Дніпро');
![alt text](image-26.png)

-- №2. IN
-- Працівники з певними ID
SELECT *
FROM employees
WHERE employee_id IN (1, 2, 3);
![alt text](image-27.png)

-- №3. BETWEEN
-- Товари з ціною від 10000 до 30000 грн
SELECT *
FROM products
WHERE unit_price BETWEEN 10000 AND 30000;
![alt text](image-28.png)

-- №4. BETWEEN
-- Замовлення за період з 15 січня до 20 серпня 2024 року
SELECT *
FROM orders
WHERE order_date BETWEEN '2024-01-15' AND '2024-08-20';
![alt text](image-29.png)

-- №5. IS NULL
-- Замовлення, для яких ще не вказана дата відправлення
SELECT *
FROM orders
WHERE shipped_date IS NULL;
![alt text](image-30.png)

-- №6. IS NOT NULL
-- Замовлення, для яких дата відправлення вже вказана
SELECT *
FROM orders
WHERE shipped_date IS NOT NULL;
![alt text](image-31.png)

#### Комбінування умов
-- №1. LIKE + AND
-- Товари Samsung, ціна яких менша за 50000 грн
SELECT *
FROM products
WHERE product_name LIKE '%Samsung%'
  AND unit_price < 50000;
![alt text](image-32.png)

-- №2. BETWEEN + IN
-- Товари з ціною від 10000 до 30000 грн
-- з певних категорій
SELECT *
FROM products
WHERE unit_price BETWEEN 10000 AND 30000
  AND category_id IN (1, 2, 3);
![alt text](image-33.png)

-- №3. IN + NOT
-- Клієнти з Києва, Львова або Одеси,
-- крім юридичних осіб
SELECT *
FROM customers
WHERE city IN ('Київ', 'Львів', 'Одеса')
  AND NOT customer_type = 'company';
![alt text](image-34.png)

-- №4. OR + IS NOT NULL
-- Замовлення, які доставлені
-- або мають вказану дату відправлення
SELECT *
FROM orders
WHERE order_status = 'delivered'
   OR shipped_date IS NOT NULL;
![alt text](image-35.png)

-- №5. LIKE + OR + IN
-- Клієнти з Києва або Львова,
-- які мають Gmail або Outlook
SELECT *
FROM customers
WHERE (email LIKE '%@gmail.com' OR email LIKE '%@outlook.com')
  AND city IN ('Київ', 'Львів');
![alt text](image-36.png)

#### Складне сортування та пагінація
-- №1. Сортування товарів за ціною,
-- а при однаковій ціні — за кількістю на складі
SELECT *
FROM products
ORDER BY unit_price DESC, units_in_stock DESC;
![alt text](image-37.png)

-- №2. Сортування клієнтів за містом,
-- а в межах міста — за іменем контакту
SELECT *
FROM customers
ORDER BY city ASC, contact_name ASC;
![alt text](image-38.png)

-- №3. Сортування замовлень за датою,
-- а при однаковій даті — за вартістю доставки
SELECT *
FROM orders
ORDER BY order_date DESC, freight DESC;
![alt text](image-39.png)

-- №4. Друга сторінка товарів по 10 записів
-- Пропускаємо перші 10 товарів
SELECT *
FROM products
ORDER BY product_id ASC
LIMIT 10 OFFSET 10;
![alt text](image-40.png)

-- №5. Третя сторінка товарів по 10 записів
-- Пропускаємо перші 20 товарів
SELECT *
FROM products
ORDER BY product_id ASC
LIMIT 10 OFFSET 20;
